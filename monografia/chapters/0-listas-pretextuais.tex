\begin{abrev}
  \item[ABNT] Associa\c{c}\~ao Brasileira de Normas T\'ecnicas
  \item[ADR] \emph{Architecture Decision Record} (registro de decis\~ao de arquitetura)
  \item[API] \emph{Application Programming Interface} (interface de programa\c{c}\~ao de aplica\c{c}\~oes)
  \item[BSD] \emph{Bidirectional Scattering Distribution} (distribui\c{c}\~ao bidirecional de espalhamento)
  \item[CAM] \emph{Computer-Aided Manufacturing} (manufatura assistida por computador)
  \item[CAD] \emph{Computer-Aided Design} (projeto assistido por computador)
  \item[CA] Crit\'erio de aceita\c{c}\~ao
  \item[DCC] \emph{Digital Content Creation} (cria\c{c}\~ao de conte\'udo digital)
  \item[DRC] \emph{Design Rule Check} (verifica\c{c}\~ao de regras de projeto)
  \item[EEVEE] \emph{Extra Easy Virtual Environment Engine} (motor de render do Blender)
  \item[FR] Requisito funcional
  \item[GLTF] \emph{Graphics Language Transmission Format}
  \item[GPU] \emph{Graphics Processing Unit} (unidade de processamento gr\'afico)
  \item[HDR] \emph{High Dynamic Range} (alta faixa dinâmica)
  \item[IOR] \emph{Index of Refraction} (indice de refra\c{c}\~ao)
  \item[JSON] \emph{JavaScript Object Notation}
  \item[LLM] \emph{Large Language Model} (modelo de linguagem de grande porte)
  \item[MCP] \emph{Model Context Protocol}
  \item[NURBS] \emph{Non-Uniform Rational B-Splines}
  \item[NFR] Requisito n\~ao funcional
  \item[PBR] \emph{Physically Based Rendering} (renderiza\c{c}\~ao fisicamente baseada)
  \item[PNG] \emph{Portable Network Graphics}
  \item[RPC] \emph{Remote Procedure Call} (chamada de procedimento remoto)
  \item[SDD] \emph{Specification-Driven Development} (desenvolvimento orientado por especifica\c{c}\~ao)
  \item[sRGB] \emph{standard RGB} (espa\c{c}o de cor padr\~ao)
  \item[UI] \emph{User Interface} (interface de usu\'ario)
  \item[UV] Mapeamento de coordenadas de textura
\end{abrev}

\begin{simb}
  \item[$w$] meia-largura m\'axima da se\c{c}\~ao transversal (m)

  \item[$z_b$] altura do fundo da se\c{c}\~ao (m)

  \item[$z_t$] altura do centro do topo (m)
  \item[$n$] n\'umero de v\'ertices de uma face
  \item[$\theta$] \^angulo de incid\^encia ou de visada (rad)
  \item[$f_0$] reflet\^ancia em incid\^encia normal
  \item[$\alpha$] rugosidade de microfacetas adimensional
  \item[$F$] termo de Fresnel
  \item[$G$] termo de sombreamento geom\'etrico
  \item[$D$] fun\c{c}\~ao de distribui\c{c}\~ao de normais
  \item[$L_o$] radi\^ancia de sa\'ida (W\,sr$^{-1}$\,m$^{-2}$)
  \item[$L_i$] radi\^ancia incidente (W\,sr$^{-1}$\,m$^{-2}$)
  \item[$\rho$] fun\c{c}\~ao de distribui\c{c}\~ao de reflet\^ancia bidirecional
\end{simb}
